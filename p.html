function getEquipos(string $filtroNombre = "", string $orden = "id_equipo") : array{
    $db = getConnection();
    $equipos = [];
    $sql = "SELECT id_equipo, nombre, email, num_integrantes, puntuacion FROM equipos 
            WHERE nombre LIKE '%$filtroNombre%' ORDER BY $orden";

    try {
        $statement = $db->prepare($sql);
        $statement->execute();
        $equipos = $statement->fetchAll(PDO::FETCH_ASSOC);
    } catch (PDOException $e) {
        die("ERROR SQL: " . $e->getMessage());
    } finally{
        $db = null;
        $statement = null;
    }
    return $equipos;
}

function getEquipoPorId($id) {
    $db = getConnection();
    $equipo = null;
    $sql = "SELECT id_equipo, nombre, email, num_integrantes, puntuacion FROM equipos WHERE id_equipo = $id";

    try {
        $statement = $db->prepare($sql);
        $statement->execute();
        $equipo = $statement->fetch(PDO::FETCH_ASSOC);
    } catch (PDOException $e) {
        die("ERROR SQL AL BUSCAR ID: " . $e->getMessage()); 
    } finally {
        $db = null;
        $statement = null;
    }
    return $equipo;
}

function actualizarEquipo($id, $nombre, $email, $num, $puntuacion) {
    $db = getConnection();
    $sql = "UPDATE equipos SET 
                nombre = '$nombre', 
                email = '$email', 
                num_integrantes = $num, 
                puntuacion = $puntuacion 
            WHERE id_equipo = $id";

    try {
        $statement = $db->prepare($sql);
        $statement->execute();
    } catch (PDOException $e) {
        die("ERROR SQL AL ACTUALIZAR: " . $e->getMessage()); 
    } finally {
        $db = null;
        $statement = null;
    }
}
